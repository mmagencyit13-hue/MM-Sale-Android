name: Build or publish MMA Tradelink APK
on:
  workflow_dispatch:
    inputs:
      publish:
        description: Publish the APK to the employee update service
        type: boolean
        default: false
concurrency:
  group: mma-android-release
  cancel-in-progress: false
permissions:
  contents: read
jobs:
  apk:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: npm
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 21
      - uses: android-actions/setup-android@v3
        with:
          packages: platform-tools
      - run: npm ci
      - run: npm test
      - run: npm run android:init
      - run: chmod +x android/gradlew
      - run: ./android/gradlew -p android assembleDebug
      - uses: actions/upload-artifact@v4
        with:
          name: MMA-Tradelink-APK
          path: android/app/build/outputs/apk/debug/app-debug.apk
          if-no-files-found: error
      - name: Publish APK to employee update service
        if: inputs.publish
        env:
          MMA_SUPABASE_SERVICE_ROLE_KEY: ${{ secrets.MMA_SUPABASE_SERVICE_ROLE_KEY }}
        run: node scripts/publish-update.mjs
